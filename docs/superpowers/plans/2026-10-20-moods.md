```kotlin
package cc.zohn.tvlauncher

import java.time.Instant
import java.time.LocalDate
import java.time.ZoneId
import java.time.ZoneOffset
import kotlin.math.acos
import kotlin.math.asin
import kotlin.math.cos
import kotlin.math.pow
import kotlin.math.roundToLong
import kotlin.math.sin
import kotlin.math.tan

// Kansas City, MO — drives the moods (sunrise/sunset).
const val HOME_LATITUDE = 39.10
const val HOME_LONGITUDE = -94.58

data class SunTimes(val sunrise: Instant, val sunset: Instant)

/** NOAA solar equations: about a minute of accuracy at mid latitudes, no network needed. */
object SunCalc {
    fun sunTimes(date: LocalDate, latitude: Double, longitude: Double, zone: ZoneId): SunTimes {
        val solar = solar(date.atTime(12, 0).atZone(zone).toInstant())
        val lat = Math.toRadians(latitude)
        // 90.833° = sun's center 50' below the horizon (refraction + radius).
        val cosHourAngle = cos(Math.toRadians(90.833)) / (cos(lat) * cos(solar.declination)) -
            tan(lat) * tan(solar.declination)
        val hourAngleDeg = Math.toDegrees(acos(cosHourAngle.coerceIn(-1.0, 1.0)))
        val utcMidnight = date.atStartOfDay(ZoneOffset.UTC).toInstant()
        val solarNoonMinutes = 720 - 4 * longitude - solar.equationOfTime
        return SunTimes(
            sunrise = utcMidnight.plusSeconds(((solarNoonMinutes - 4 * hourAngleDeg) * 60).roundToLong()),
            sunset = utcMidnight.plusSeconds(((solarNoonMinutes + 4 * hourAngleDeg) * 60).roundToLong()),
        )
    }

    /** Geometric sun elevation in degrees (negative = below the horizon). */
    fun elevationDegrees(instant: Instant, latitude: Double, longitude: Double): Double {
        val solar = solar(instant)
        val utcMinutes = instant.epochSecond.mod(86_400L) / 60.0
        val trueSolarMinutes = (utcMinutes + solar.equationOfTime + 4 * longitude).mod(1440.0)
        val hourAngle = Math.toRadians(
            if (trueSolarMinutes / 4 < 0) trueSolarMinutes / 4 + 180 else trueSolarMinutes / 4 - 180
        )
        val lat = Math.toRadians(latitude)
        val cosZenith = sin(lat) * sin(solar.declination) +
            cos(lat) * cos(solar.declination) * cos(hourAngle)
        return 90 - Math.toDegrees(acos(cosZenith.coerceIn(-1.0, 1.0)))
    }

    private class Solar(val declination: Double, val equationOfTime: Double) // radians, minutes

    private fun solar(instant: Instant): Solar {
        val julianDay = instant.epochSecond / 86_400.0 + 2_440_587.5
        val t = (julianDay - 2_451_545.0) / 36_525.0
        val meanLong = (280.46646 + t * (36_000.76983 + t * 0.0003032)).mod(360.0)
        val meanAnomaly = Math.toRadians(357.52911 + t * (35_999.05029 - 0.0001537 * t))
        val eccentricity = 0.016708634 - t * (0.000042037 + 0.0000001267 * t)
        val center = sin(meanAnomaly) * (1.914602 - t * (0.004817 + 0.000014 * t)) +
            sin(2 * meanAnomaly) * (0.019993 - 0.000101 * t) + sin(3 * meanAnomaly) * 0.000289
        val omega = Math.toRadians(125.04 - 1_934.136 * t)
        val apparentLong = Math.toRadians(meanLong + center - 0.00569 - 0.00478 * sin(omega))
        val meanObliquity = 23 + (26 + (21.448 - t * (46.815 + t * (0.00059 - t * 0.001813))) / 60) / 60
        val obliquity = Math.toRadians(meanObliquity + 0.00256 * cos(omega))
        val declination = asin(sin(obliquity) * sin(apparentLong))
        val y = tan(obliquity / 2).pow(2)
        val l0 = Math.toRadians(meanLong)
        val equationOfTime = 4 * Math.toDegrees(
            y * sin(2 * l0) - 2 * eccentricity * sin(meanAnomaly) +
                4 * eccentricity * y * sin(meanAnomaly) * cos(2 * l0) -
                0.5 * y * y * sin(4 * l0) - 1.25 * eccentricity * eccentricity * sin(2 * meanAnomaly)
        )
        return Solar(declination, equationOfTime)
    }
}
```

- [ ] **Step 3: Implement `Mood.kt`**

Note: the spec says the clock "sleeps until the next window"; this plan caps sleeps at 10 minutes so Zohn's sun keeps moving across the sky between blends (re-rendering the scene every 10 min is cheap).

```kotlin
package cc.zohn.tvlauncher

import java.time.Duration
import java.time.Instant
import java.time.LocalDate
import java.time.ZoneId

enum class Phase {
    MORNING, DAY, EVENING, NIGHT;

    fun next(): Phase = entries[(ordinal + 1) % entries.size]
}

/**
 * Where we are in the day. Outside a blend window: from = current phase, to = the next one,
 * blend = 0. Inside a 45-minute window: blend runs 0→1 from [from] to [to]; the phase
 * officially switches at the window's midpoint.
 */
data class MoodState(val from: Phase, val to: Phase, val blend: Float, val sunElevation: Double) {
    val phase: Phase get() = if (blend >= 0.5f) to else from
}

object Mood {
    val BLEND: Duration = Duration.ofMinutes(45)
    private val HALF: Duration = BLEND.dividedBy(2)
    private val BLEND_TICK: Duration = Duration.ofMinutes(2)
    private val MAX_SLEEP: Duration = Duration.ofMinutes(10)

    /** Phase starts (blend midpoints) for one day's sun times. */
    fun boundaries(sun: SunTimes): List<Pair<Instant, Phase>> = listOf(
        sun.sunrise.minus(Duration.ofMinutes(30)) to Phase.MORNING,
        sun.sunrise.plus(Duration.ofHours(3)) to Phase.DAY,
        sun.sunset.minus(Duration.ofHours(1)) to Phase.EVENING,
        sun.sunset.plus(Duration.ofMinutes(150)) to Phase.NIGHT,
    )

    /** Yesterday's, today's and tomorrow's boundaries in order, so night can span midnight. */
    fun boundariesAround(date: LocalDate, zone: ZoneId): List<Pair<Instant, Phase>> =
        listOf(date.minusDays(1), date, date.plusDays(1)).flatMap {
            boundaries(SunCalc.sunTimes(it, HOME_LATITUDE, HOME_LONGITUDE, zone))
        }

    fun stateAt(now: Instant, boundaries: List<Pair<Instant, Phase>>, sunElevation: Double): MoodState {
        val lastIndex = boundaries.indexOfLast { !it.first.isAfter(now) }
        require(lastIndex in 1 until boundaries.lastIndex) { "boundaries must surround $now" }
        val (lastAt, current) = boundaries[lastIndex]
        val (nextAt, upcoming) = boundaries[lastIndex + 1]
        val previous = boundaries[lastIndex - 1].second
        return when {
            now.isBefore(lastAt.plus(HALF)) ->
                MoodState(previous, current, fraction(lastAt.minus(HALF), now), sunElevation)
            !now.isBefore(nextAt.minus(HALF)) ->
                MoodState(current, upcoming, fraction(nextAt.minus(HALF), now), sunElevation)
            else -> MoodState(current, upcoming, 0f, sunElevation)
        }
    }

    /** 2 min inside a blend window; otherwise until the next window opens, at most 10 min. */
    fun nextTickDelay(now: Instant, boundaries: List<Pair<Instant, Phase>>): Duration {
        val inWindow = boundaries.any { (at, _) -> !now.isBefore(at.minus(HALF)) && now.isBefore(at.plus(HALF)) }
        if (inWindow) return BLEND_TICK
        val nextWindow = boundaries.map { it.first.minus(HALF) }.firstOrNull { it.isAfter(now) }
            ?: return MAX_SLEEP
        val untilWindow = Duration.between(now, nextWindow)
        return if (untilWindow < MAX_SLEEP) untilWindow else MAX_SLEEP
    }

    private fun fraction(windowStart: Instant, now: Instant): Float =
        (Duration.between(windowStart, now).toMillis().toFloat() / BLEND.toMillis()).coerceIn(0f, 1f)
}
```

```kotlin
package cc.zohn.tvlauncher

import android.os.SystemClock
import kotlinx.coroutines.delay
import java.time.Instant
import java.time.ZoneId

private const val FAST_DAY_SPEEDUP = 720L // one day in two minutes
private const val FAST_DAY_TICK_MS = 1_000L

/**
 * Ticks the current [MoodState]: every 2 min inside a blend, otherwise at most every 10 min.
 * Restart [run] (cancel + relaunch) after a time/zone change or a fast-day toggle.
 */
class MoodClock(private val zone: () -> ZoneId = { ZoneId.systemDefault() }) {
    var fastDay = false
        private set
    private var fastStartRealtime = 0L
    private var fastStartInstant: Instant = Instant.EPOCH

    // Debug: plays a whole day in two minutes, starting from now.
    fun setFastDay(on: Boolean) {
        fastDay = on
        fastStartRealtime = SystemClock.elapsedRealtime()
        fastStartInstant = Instant.now()
    }

    fun now(): Instant = if (!fastDay) Instant.now() else
        fastStartInstant.plusMillis((SystemClock.elapsedRealtime() - fastStartRealtime) * FAST_DAY_SPEEDUP)

    fun state(): MoodState = stateAt(now())

    suspend fun run(onState: (MoodState) -> Unit) {
        while (true) {
            val now = now()
            onState(stateAt(now))
            val wait = if (fastDay) FAST_DAY_TICK_MS
            else Mood.nextTickDelay(now, boundaries(now)).toMillis().coerceAtLeast(1_000L)
            delay(wait)
        }
    }

    private fun boundaries(now: Instant) = Mood.boundariesAround(now.atZone(zone()).toLocalDate(), zone())

    private fun stateAt(now: Instant) = Mood.stateAt(
        now,
        boundaries(now),
        SunCalc.elevationDegrees(now, HOME_LATITUDE, HOME_LONGITUDE),
    )
}
```

```kotlin
/**
 * Zohn's sky, sun, stars, horizon and ground, drawn on the CPU once per mood tick (never per
 * frame). [horizonFraction] is where the hero/dock divider sits, as a fraction of screen height.
 */
fun renderSunsetScene(p: LauncherPalette, sunElevation: Double, horizonFraction: Float): ImageBitmap {
    val bitmap = ImageBitmap(SCENE_W, SCENE_H)
    CanvasDrawScope().draw(Density(1f), LayoutDirection.Ltr, Canvas(bitmap), Size(SCENE_W.toFloat(), SCENE_H.toFloat())) {
        val w = size.width
        val horizonY = size.height * horizonFraction
        val sky = Brush.verticalGradient(
            0f to p.sky1, 0.67f to p.sky2, 0.9f to p.sky3, 1f to p.horizon,
            startY = 0f, endY = horizonY,

        if (p.starAlpha > 0f) {
            val random = Random(7)
            repeat(STAR_COUNT) {
                val x = random.nextFloat() * w
                val y = random.nextFloat() * horizonY * 0.85f
                drawCircle(Color.White.copy(alpha = p.starAlpha * (0.4f + 0.6f * random.nextFloat())), 0.8f + random.nextFloat(), Offset(x, y))
            }
        }

        // Sun: height follows real elevation; striped lower half; clipped at the horizon.
        if (sunElevation > SUN_HIDDEN_BELOW) {
            val radius = w * SUN_RADIUS
            val lift = (sunElevation / SUN_FULL_ELEVATION).coerceIn(-0.3, 1.0).toFloat()
            val center = Offset(w * SUN_CENTER_X, horizonY - lift * horizonY * 0.55f)
            clipRect(bottom = horizonY) {
                val sunBrush = Brush.verticalGradient(
                    0f to p.sunTop, 0.6f to p.sunMid, 1f to p.sunBottom,
                    startY = center.y - radius, endY = center.y + radius,
                )
                drawCircle(sunBrush, radius, center)
                // Stripe gaps are painted with the sky brush, identical to the sky behind them.
                val stripes = listOf(0.45f to 0.49f, 0.58f to 0.62f, 0.70f to 0.74f, 0.81f to 0.86f)
                val circle = Path().apply { addOval(Rect(center, radius)) }
                clipPath(circle) {
                    for ((from, to) in stripes) {
                        val top = center.y - radius + 2 * radius * from
                        val bottom = center.y - radius + 2 * radius * to
                        drawRect(sky, topLeft = Offset(center.x - radius, top), size = Size(2 * radius, bottom - top))
                    }
                }
            }
        }

        // Ground + horizon line (this is the divider between hero and dock in Zohn's profile).
        drawRect(p.ground, topLeft = Offset(0f, horizonY), size = Size(w, size.height - horizonY))
        drawLine(p.horizon, Offset(0f, horizonY), Offset(w, horizonY), strokeWidth = 2f)

        // Soft dark veil behind the hero text.
        drawRect(
            Brush.radialGradient(
                listOf(p.ground.copy(alpha = 0.5f), Color.Transparent),
                center = Offset(w * 0.22f, size.height * 0.5f),
                radius = w * 0.4f,
            )
        )
    }
    return bitmap
}

/**
 * Perspective grid below the horizon, drawn as ~36 lines per frame (no full-screen fill).
 * Scrolls toward the viewer while the remote is idle; frozen while navigating and when
 * [animate] is false (night).
 */
@Composable
fun GridFloor(p: LauncherPalette, horizonFraction: Float, animate: Boolean, lastInputAt: () -> Long) {
    val offset = remember { Animatable(0f) }
    if (animate) LaunchedEffect(Unit) { scrollWhileIdle(offset, lastInputAt) }

    Canvas(Modifier.fillMaxSize()) {
        if (p.gridAlpha <= 0f) return@Canvas
        val horizonY = size.height * horizonFraction
        val depth = size.height - horizonY
        // Horizontal lines bunch up toward the horizon and fade into it.
        for (i in 0..GRID_LINES) {
            val t = ((i + offset.value) / GRID_LINES).coerceIn(0f, 1f)
            val y = horizonY + depth * t * t
            drawLine(p.gridH.copy(alpha = p.gridH.alpha * p.gridAlpha * t), Offset(0f, y), Offset(size.width, y), strokeWidth = 1.5f)
        }
        // Vertical lines converge on a vanishing point at the horizon's center.
        val vanishing = Offset(size.width / 2, horizonY)
        val columnBrush = Brush.verticalGradient(
            listOf(Color.Transparent, p.gridV.copy(alpha = p.gridV.alpha * p.gridAlpha)),
            startY = horizonY, endY = size.height,
        )
        for (i in -GRID_COLUMNS / 2..GRID_COLUMNS / 2) {
            val bottomX = size.width / 2 + i * size.width * 2f / GRID_COLUMNS
            drawLine(columnBrush, vanishing, Offset(bottomX, size.height), strokeWidth = 1.5f)
        }
    }
}

private suspend fun scrollWhileIdle(offset: Animatable<Float, *>, lastInputAt: () -> Long) {
    while (true) {
        val idleFor = SystemClock.uptimeMillis() - lastInputAt()
        if (idleFor < GRID_IDLE_MS) {
            delay(GRID_IDLE_MS - idleFor)
            continue
        }
        coroutineScope {
            val scroll = launch {
                while (true) {
                    offset.animateTo(1f, tween(((1f - offset.value) * GRID_STEP_MS).toInt().coerceAtLeast(1), easing = LinearEasing))
                    offset.snapTo(0f)
                }
            }
            val watcher = launch {
                snapshotFlow { lastInputAt() }.drop(1).first()
                scroll.cancel()
            }
            scroll.join()
            watcher.cancel()
        }
    }
}
```

```kotlin
        val today = LocalDate.now().toEpochDay()
```

```kotlin
package cc.zohn.tvlauncher

import android.content.Context
import android.util.Log
import java.io.File
import java.time.OffsetDateTime
import java.time.ZonedDateTime
import java.time.format.DateTimeFormatter

private const val TAG = "TvLauncher"
private const val KEEP_DAYS = 90L

/** Launch history for the future Habits feature. Stays on the TV; nothing reads it yet. */
object UsageLog {
    fun line(at: ZonedDateTime, phase: Phase, profileId: String, packageName: String): String =
        listOf(
            at.toOffsetDateTime().format(DateTimeFormatter.ISO_OFFSET_DATE_TIME),
            at.dayOfWeek.name,
            at.hour.toString(),
            phase.name,
            profileId,
            packageName,
        ).joinToString(",")

    fun trim(lines: List<String>, now: ZonedDateTime): List<String> {
        val cutoff = now.minusDays(KEEP_DAYS).toOffsetDateTime()
        return lines.filter { line ->
            runCatching { OffsetDateTime.parse(line.substringBefore(',')) }.getOrNull()?.isAfter(cutoff) == true
        }
    }

    fun append(context: Context, line: String) {
        try {
            file(context).appendText(line + "\n")
        } catch (e: Exception) {
            Log.w(TAG, "usage log write failed", e)
        }
    }

    fun trimFile(context: Context, now: ZonedDateTime) {
        try {
            val file = file(context)
            if (file.exists()) file.writeText(trim(file.readLines(), now).joinToString("") { it + "\n" })
        } catch (e: Exception) {
            Log.w(TAG, "usage log trim failed", e)
        }
    }

    private fun file(context: Context) = File(context.filesDir, "usage.log")
}
```
